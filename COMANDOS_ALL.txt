git pull origin develop

git status

------------------------------COMANDOS KAFKA-----------------------------------------------------------------------------
-------------------------------------------------------------------------------------------------------------------------
-Descargar https://kafka.apache.org/community/downloads/
De la opcion  Binary downloads: Scala 2.12 

netstat -aon | findstr :8081
taskkill /pid numPid /F
kafka_manager.bat > log_kafka_info.txt
cd C:\kafka_2.11-2.0.0
inicia zookeeper
.\bin\windows\zookeeper-server-start.bat .\config\zookeeper.properties

Iniciar Apache Kafka
.\bin\windows\kafka-server-start.bat .\config\server.properties

Listar topicos
.\bin\windows\kafka-topics.bat --list --zookeeper localhost:2181

Describir topico
.\bin\windows\kafka-topics.bat --describe --zookeeper localhost:2181

borrar topico
.\bin\windows\kafka-topics.bat --zookeeper localhost:2181 --delete --topic sol

crear topico
.\bin\windows\kafka-topics.bat --create --zookeeper localhost:2181 --replication-factor 1 --partitions 1 --topic sol
.\bin\windows\kafka-topics.bat --create --zookeeper localhost:2181 --replication-factor 1 --partitions 1 --topic respuesta
.\bin\windows\kafka-topics.bat --create --zookeeper localhost:2181 --replication-factor 1 --partitions 1 --topic canal

Crear productor
.\bin\windows\kafka-console-producer.bat --broker-list localhost:9092 --topic sol
.\bin\windows\kafka-console-producer.bat --broker-list localhost:9092 --topic respuesta
.\bin\windows\kafka-console-producer.bat --broker-list localhost:9092 --topic canal

Crear consumidor lista todos los mensajes
.\bin\windows\kafka-console-consumer.bat --bootstrap-server localhost:9092 --topic sol --from-beginning

solo recibe los mensajes recientes
.\bin\windows\kafka-console-consumer.bat --bootstrap-server localhost:9092 --topic sol
.\bin\windows\kafka-console-consumer.bat --bootstrap-server localhost:9092 --topic respuesta
.\bin\windows\kafka-console-consumer.bat --bootstrap-server localhost:9092 --topic canal

------------------------------COMANDOS DOCKER-----------------------------------------------------------------------------
-------------------------------------------------------------------------------------------------------------------------
Para importar una imagen de docker hub sitio web
docker pull mysql:9.0

Ubicamos el dockerfile y ejecutamos ponemos el nombre a la imagen
docker build -t "imagen-app-docker" .
docker run --name personas-springboot-docker -p 8084:8084 personas-docker:latest

lista imagenes
docker images

borra la imagen
docker rmi name_image

Para listar contenedores
docker ps -a
docker rm personas-springboot-docker
FORZADO BORRADO
docker rm -f personas-springboot-docker

ubicamos en la ruta de docker-compose.yml y ejecutamos
docker compose up -d
en win 7
docker-compose up -d

Para acceder al logs del contenedor
docker logs employee
